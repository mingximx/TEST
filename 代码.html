        let lockedSpots = new Set();

        let currentSpot = null;


        // 初始化

        function init() {

            const grid = document.getElementById('gridContainer');

            const saved = localStorage.getItem('veniceCosLocked');

            if (saved) {

                lockedSpots = new Set(JSON.parse(saved));

            }


            for (let i = 1; i <= 100; i++) {

                const card = document.createElement('div');

                card.className = `number-card ${lockedSpots.has(i) ? 'locked' : 'available'}`;

                card.textContent = i.toString().padStart(2, '0');

                card.onclick = () => openModal(i);

                

                if (lockedSpots.has(i)) {

                    const spotName = document.createElement('span');

                    spotName.className = 'spot-name';

                    spotName.textContent = spots[i-1].name;

                    card.appendChild(spotName);

                }

                

                grid.appendChild(card);

            }


            updateStats();

        }


        function openModal(num) {

            if (lockedSpots.has(num)) return;

            

            currentSpot = num;

            const spot = spots[num-1];

            document.getElementById('modalNumber').textContent = num.toString().padStart(2, '0');

            document.getElementById('modalSpot').textContent = spot.name;

            document.getElementById('modalDesc').textContent = spot.desc;

            document.getElementById('modal').classList.add('active');

        }


        function closeModal() {

            document.getElementById('modal').classList.remove('active');

            currentSpot = null;

        }


        function confirmLock() {

            if (!currentSpot) return;

            

            lockedSpots.add(currentSpot);

            localStorage.setItem('veniceCosLocked', JSON.stringify([...lockedSpots]));

            

            // 更新UI

            const cards = document.querySelectorAll('.number-card');

            const card = cards[currentSpot - 1];

            card.className = 'number-card locked';

            card.onclick = null;

            

            const spotName = document.createElement('span');

            spotName.className = 'spot-name';

            spotName.textContent = spots[currentSpot-1].name;

            card.appendChild(spotName);

            

            // 庆祝动画

            const cel = document.getElementById('celebration');

            cel.classList.add('show');

            setTimeout(() => cel.classList.remove('show'), 1000);

            

            updateStats();

            closeModal();

            

            // 检查是否完成

            if (lockedSpots.size === 100) {

                setTimeout(() => alert('🎉 恭喜！你已打卡威尼斯全部100个景点！\n\nCOS环球记威尼斯篇，期待你的正片！'), 300);

            }

        }


        function updateStats() {

            const locked = lockedSpots.size;

            const percent = Math.round(locked);

            

            document.getElementById('lockedCount').textContent = locked;

            document.getElementById('availableCount').textContent = 100 - locked;

            document.getElementById('percentCount').textContent = percent + '%';

            document.getElementById('progressFill').style.width = percent + '%';

        }


        function resetAll() {

            if (lockedSpots.size === 0) return;

            

            if (confirm(`确定要重置所有打卡记录吗？\n\n已打卡 ${lockedSpots.size} 个景点将全部解锁，不可恢复。`)) {

                lockedSpots.clear();

                localStorage.removeItem('veniceCosLocked');

                

                const grid = document.getElementById('gridContainer');

                grid.innerHTML = '';

                init();

            }

        }


        // 点击模态框外部关闭

        document.getElementById('modal').onclick = function(e) {

            if (e.target === this) closeModal();

        };


        // 初始化

        init();

    </script>
